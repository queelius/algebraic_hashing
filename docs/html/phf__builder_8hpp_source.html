124  auto operator()(I begin, I end)
125  {

129  auto N = (size_t)std::ceil(m/r);

131  size_t l_star;

133  size_t collides_star = 0;
134  std::mutex lck;
135  auto const start_time = std::chrono::system_clock::now();
136 
137  auto task = [&](size_t start, size_t stop)
138  {
139  std::vector<bool> K(N);
140  for (auto l = start; l != stop; ++l)
141  {
142  auto const cur_time = std::chrono::system_clock::now();
143  auto const elapsed = std::chrono::duration_cast<std::chrono::milliseconds>(
144  cur_time - start_time);
145  if (succ_star == m || elapsed > duration)
146  break;
147  size_t succ = 0;
148  size_t collides = 0;
149  for (auto x = begin; x != end; ++x)
150  {
151  auto hash = (h(*x) ^ l) % N;
152  if (K[hash])
153  {
154  if (++collides > collides_star)
155  break;
156  }
157  else
158  {
159  ++succ;
160  K[hash] = true;
161  }
162  }
163 
164  if (succ > succ_star)
165  {
166  lck.lock();
167  if (succ > succ_star)
168  {
169  l_star = l;
170  succ_star = succ;
171  collides_star = collides;
172  }
173  lck.unlock();
174  }
175  K.assign(N,false);
176  }
177  };
