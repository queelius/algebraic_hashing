116 // 64-bit FNV-1a implementation
117 template<concepts::Hashable T>
118 constexpr hash64 hash_dispatch(T const& input, tag_64) const {
119 std::uint64_t hash = constants::offset_basis_64;
120 auto bytes = get_byte_representation(input);
121
122 for (std::uint8_t byte : bytes) {
123 hash ^= static_cast<std::uint64_t>(byte);
124 hash *= constants::prime_64;
125 }
126
127 // Convert to our hash_value type
128 hash64 result;
129 std::memcpy(result.data().data(), &hash, sizeof(hash));
130 return result;
131 }

172 template<typename T>
173 constexpr auto get_byte_representation(T const& input) const {
174 if constexpr (std::is_arithmetic_v<T>) {
175 // Handle fundamental types by direct byte access
176 return get_arithmetic_bytes(input);
177 } else if constexpr (std::is_array_v<T> && std::same_as<std::remove_extent_t<T>, char>) {
178 // Handle const char arrays (string literals)
179 auto len = std::extent_v<T> > 0 ? std::extent_v<T> - 1 : 0; // Exclude null terminator
180 return std::span<const std::uint8_t>(
181 reinterpret_cast<const std::uint8_t*>(input),
182 len
183 );
184 } else if constexpr (std::same_as<T, std::string> || std::same_as<T, std::string_view>) {
185 // String types
186 return std::span<const std::uint8_t>(
187 reinterpret_cast<const std::uint8_t*>(input.data()),
188 input.size()
189 );
190 } else if constexpr (std::ranges::contiguous_range<T>) {
191 // Handle containers like vector, array
192 return get_container_bytes(input);
193 } else if constexpr (requires { std::tuple_size_v<T>; }) {
194 // Handle tuples
195 return get_tuple_bytes(input);
196 } else {
197 // Fallback: treat as raw memory
198 return std::span<const std::uint8_t>(
199 reinterpret_cast<const std::uint8_t*>(&input),
200 sizeof(T)
201 );
202 }
203 }

205 template<std::integral T>
206 constexpr auto get_arithmetic_bytes(T value) const {
207 // Convert to little-endian byte array for platform independence
208 std::array<std::uint8_t, sizeof(T)> bytes;
209 for (std::size_t i = 0; i < sizeof(T); ++i) {
210 bytes[i] = static_cast<std::uint8_t>(value & 0xFF);
211 value >>= 8;
212 }
213 // Return array directly, not a span to avoid stack-use-after-return
214 return bytes;
215 }
216
217 template<std::floating_point T>
218 constexpr auto get_arithmetic_bytes(T value) const {
219 // For floating point, we use bit_cast for deterministic representation
220 if constexpr (sizeof(T) == sizeof(std::uint32_t)) {
221 auto int_value = std::bit_cast<std::uint32_t>(value);
222 return get_arithmetic_bytes(int_value);
223 } else if constexpr (sizeof(T) == sizeof(std::uint64_t)) {
224 auto int_value = std::bit_cast<std::uint64_t>(value);
225 return get_arithmetic_bytes(int_value);
226 } else {
227 // Fallback for unusual float types - copy to array
228 std::array<std::uint8_t, sizeof(T)> bytes;
229 std::memcpy(bytes.data(), &value, sizeof(T));
230 return bytes;
231 }
232 }

264 // ==================== Hash Extension for Large Outputs ====================
265
266 HashType extend_hash(hash64 const& base_hash) const {
267 static_assert(sizeof(HashType) > sizeof(hash64));
268
269 HashType result;
270 auto result_bytes = result.bytes();
271
272 // Copy base hash
273 std::ranges::copy(base_hash.bytes(), result_bytes.begin());
274
275 // Fill remaining bytes by rehashing with different salts
276 std::size_t remaining_bytes = sizeof(HashType) - sizeof(hash64);
277 std::size_t offset = sizeof(hash64);
278
279 while (remaining_bytes > 0) {
280 // Create a salted version and hash the base hash
281 auto salted = base_hash ^ hash64{}; // Simplified salting
282 auto rehashed = hash_dispatch(salted, tag_64{});
283
284 std::size_t copy_bytes = std::min(remaining_bytes, sizeof(hash64));
285 std::ranges::copy_n(rehashed.bytes().begin(), copy_bytes,
286 result_bytes.begin() + offset);
287
288 remaining_bytes -= copy_bytes;
289 offset += copy_bytes;
290 }
291
292 return result;
293 }
